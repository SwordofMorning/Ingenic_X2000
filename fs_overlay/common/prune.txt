# Paths (relative to the root filesystem) that must NOT be present in the image.
#
# Why this file exists: buildroot never uninstalls a package. When a package or
# a rootfs_config option is disabled in a new product configuration, the files it
# installed earlier stay in buildroot's persistent output/target and would end up
# in the shipped image. This list removes them (before the overlay is merged), so
# a configuration change does not silently keep dead files around.
#
# One path per line; lines starting with # are comments.
# This file and extra-libs.txt are excluded from the overlay copy by build.sh.

# adb is disabled in the product defconfig (APP_br_adb_server is not set): the
# init scripts are dropped by the vendor rootfs_config logic, but the binary
# comes from the buildroot package android-tools and has to be pruned here.
usr/bin/adbd

# vendor factory-test tool (APP_test_shell is not set): the script runs an
# interactive device test at boot and the package installs its helper scripts
# under the etc directory.
etc/init.d/S99factory_test_shell
etc/factory_test

# vendor WiFi autostart: replaced by our own manager (S60WiFi), which brings the
# interface up with the stored profile and also supports AP mode.
# S41 (BT firmware download) and S43 (MAC address setup) are kept: S60WiFi calls
# S43 for the MAC and leaves Bluetooth to the vendor scripts.
etc/init.d/S44wifi_bcm_up

# vendor SSH adaptation for a read-only root filesystem (APP_br_ssh is not set):
# etc/ssh was a symlink into the data partition and /save held the real config
# which S30cp_ssh copied at boot. This product uses a plain, writable /etc/ssh
# (see fs_overlay/common/etc/ssh/sshd_config and etc/init.d/S80SSH).
etc/ssh
save
etc/init.d/S30cp_ssh

# buildroot's own start script for OpenSSH. It only tests whether the host key
# files EXIST, so a zero-length key file (seen on this board: no RTC, freshly
# seeded random pool) kept sshd dead on every boot. Replaced by our S80SSH,
# which validates the keys and repairs what it can.
etc/init.d/S50sshd
